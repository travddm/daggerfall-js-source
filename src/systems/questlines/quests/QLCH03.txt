Item _letter_ letter used 1011

Clock _wait_ 60.00:00 0 flag 1 range 0 1

--	Quest start-up:
	ql name _ysolde_ Ysolde_Marnhel
	place npc _ysolde_ at _hall_
	start timer _courier_

_courier_ task:
	give pc _letter_ notify 1012
	log 1010 step 0
	start timer _wait_

_met_ task:
	clicked npc _ysolde_
	say 1013
	ask 1030 choices 1031 _tokeep_ 1032 _toseal_

variable _tokeep_
variable _toseal_

_keepway_ task:
	when _tokeep_
	say 1040
	outcome _keep_
	give pc _gold_
	give pc _worth_
	end quest

_sealway_ task:
	when _toseal_
	say 1041
	outcome _seal_
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
